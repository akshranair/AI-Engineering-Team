# Python program to calculate the first 10,000 terms of the series
# 1 - 1/3 + 1/5 - 1/7 + ... and multiply the total by 4.

# Initialize the sum
total_sum = 0.0

# Loop through the first 10,000 terms
for i in range(10000):
    # Determine the sign based on the term index
    sign = 1 if i % 2 == 0 else -1
    # Calculate the term
    term = sign / (2 * i + 1)
    # Add the term to the total sum
    total_sum += term

# Multiply the total sum by 4
final_result = total_sum * 4

# Print the result
print(final_result)
